  • modernize · low impact

    sanitizeInput function: Replace method chaining with a single replace using regex and a lookup map for better performance and maintainability

    Before
    export function sanitizeInput(input: string): string {
  return input
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}
    After
    export function sanitizeInput(input: string): string {
  const htmlEscapes: Record<string, string> = {
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#039;'
  };
  return input.replace(/[<>"']/g, char => htmlEscapes[char]);
}

    Benefits: Reduces multiple string traversals from 4 to 1, improves performance on large inputs, and makes the escape mappings more explicit and easier to extend
