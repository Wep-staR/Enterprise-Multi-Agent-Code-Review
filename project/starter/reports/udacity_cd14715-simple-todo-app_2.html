    Before
    export async function searchTodos(keyword: string): Promise<Todo[]> {
  const query = `SELECT * FROM todos WHERE title LIKE '%${keyword}%' OR description LIKE '%${keyword}%'`;

    After
    export async function searchTodos(keyword: string): Promise<Todo[]> {

    After
